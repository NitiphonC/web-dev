2. ออกแบบ 2 ตาราง

เลือกโมเดลที่ "มีตารางลูก" ชัด ๆ เพื่อให้ได้ FK และได้คะแนนเรื่อง relation

categories

  • id PK
  • name
  • description
  • created_at
1 ──── N

books

  • id PK
  • title
  • author
  • price, stock
  • cover_url
  • category_id FK → categories.id
CREATE TABLE categories (
  id          INTEGER PRIMARY KEY AUTOINCREMENT,
  name        TEXT    NOT NULL UNIQUE,
  description TEXT    DEFAULT '',
  created_at  TEXT    NOT NULL DEFAULT (datetime('now'))
);

CREATE TABLE books (
  id          INTEGER PRIMARY KEY AUTOINCREMENT,
  title       TEXT    NOT NULL,
  author      TEXT    NOT NULL,
  price       REAL    NOT NULL DEFAULT 0,
  stock       INTEGER NOT NULL DEFAULT 0,
  cover_url   TEXT,
  category_id INTEGER NOT NULL REFERENCES categories(id) ON DELETE RESTRICT,
  created_at  TEXT    NOT NULL DEFAULT (datetime('now'))
);

PRAGMA foreign_keys = ON;

ตัวเลือกคู่ตารางอื่นที่เจอบ่อย

ตารางหลักตารางลูกไฟล์ที่อัปโหลด
bookscategories / authorsรูปปก
productscategories / ordersรูปสินค้า
courseslessons / enrollmentsสลิป / เอกสาร
userspostsavatar / แนบไฟล์

3. Backend CRUD

ทำตารางเดียวให้ครบก่อน แล้วก็อปไปแปลง — เร็วกว่าเขียนใหม่

โครง Express ขั้นต่ำ

const express = require("express");
const db = require("./db");
const app = express();

app.use(express.json());
app.use(express.static("public"));

// READ (list)
app.get("/api/books", (req, res) => {
  const rows = db.prepare(`
    SELECT b.*, c.name AS category_name
    FROM books b JOIN categories c ON c.id = b.category_id
    ORDER BY b.id DESC`).all();
  res.json({ data: rows, total: rows.length });
});

// READ (one)
app.get("/api/books/:id", (req, res) => {
  const row = db.prepare("SELECT * FROM books WHERE id = ?").get(req.params.id);
  if (!row) return res.status(404).json({ message: "ไม่พบข้อมูล" });
  res.json(row);
});

// CREATE
app.post("/api/books", (req, res) => {
  const { title, author, price = 0, stock = 0, category_id } = req.body;
  if (!title) return res.status(400).json({ message: "ต้องระบุชื่อ" });

  const info = db.prepare(
    "INSERT INTO books (title, author, price, stock, category_id) VALUES (?,?,?,?,?)"
  ).run(title, author, price, stock, category_id);

  res.status(201).json({ id: Number(info.lastInsertRowid) });
});

// UPDATE
app.put("/api/books/:id", (req, res) => {
  const { title, author, price, stock, category_id } = req.body;
  const info = db.prepare(
    "UPDATE books SET title=?, author=?, price=?, stock=?, category_id=? WHERE id=?"
  ).run(title, author, price, stock, category_id, req.params.id);

  if (info.changes === 0) return res.status(404).json({ message: "ไม่พบข้อมูล" });
  res.json({ ok: true });
});

// DELETE
app.delete("/api/books/:id", (req, res) => {
  const info = db.prepare("DELETE FROM books WHERE id = ?").run(req.params.id);
  if (info.changes === 0) return res.status(404).json({ message: "ไม่พบข้อมูล" });
  res.status(204).end();
});

app.listen(3000);

Status code ที่ควรใช้

  • 200 อ่าน/แก้สำเร็จ
  • 201 สร้างสำเร็จ
  • 204 ลบสำเร็จ (ไม่มี body)
  • 400 ข้อมูลไม่ถูกต้อง
  • 404 ไม่พบ
  • 409 ชนกัน (ชื่อซ้ำ / ลบไม่ได้)

กับดักที่เจอบ่อย

  • ลืม express.json() → req.body เป็น undefined
  • ลืม await/callback → ตอบก่อนบันทึกเสร็จ
  • id เป็น string จาก path → แปลงเป็น Number ก่อน
  • ลบ parent ที่มี child อยู่ → ต้องเช็กก่อน

4. อัปโหลดไฟล์

ท่องให้ขึ้นใจ: multer.single("ชื่อ field") ต้องตรงกับ FormData ฝั่งหน้าเว็บ

ฝั่ง server

const multer = require("multer");
const path = require("path");
const fs = require("fs");

const UPLOAD_DIR = path.join(__dirname, "uploads");
fs.mkdirSync(UPLOAD_DIR, { recursive: true });

const storage = multer.diskStorage({
  destination: (req, file, cb) => cb(null, UPLOAD_DIR),
  filename: (req, file, cb) => {
    const ext = path.extname(file.originalname).toLowerCase();
    cb(null, `${Date.now()}-${Math.random().toString(36).slice(2, 8)}${ext}`);
  },
});

const upload = multer({
  storage,
  limits: { fileSize: 2 * 1024 * 1024 },
  fileFilter: (req, file, cb) => {
    if (!file.mimetype.startsWith("image/")) return cb(new Error("ต้องเป็นรูปภาพ"));
    cb(null, true);
  },
});

// ให้เข้าถึงไฟล์ที่อัปโหลดได้
app.use("/uploads", express.static(UPLOAD_DIR));

// route ที่รับไฟล์
app.post("/api/books", upload.single("cover"), (req, res) => {
  const coverUrl = req.file ? `/uploads/${req.file.filename}` : null;
  const info = db.prepare(
    "INSERT INTO books (title, author, category_id, cover_url) VALUES (?,?,?,?)"
  ).run(req.body.title, req.body.author, req.body.category_id, coverUrl);

  res.status(201).json({ id: Number(info.lastInsertRowid), coverUrl });
});

ฝั่งหน้าเว็บ

const formData = new FormData();
formData.append("title", title);
formData.append("author", author);
formData.append("category_id", categoryId);
formData.append("cover", fileInput.files[0]); // ต้องตรงชื่อ field

const res = await fetch("/api/books", { method: "POST", body: formData });
// ห้ามใส่ Content-Type เอง ให้ browser ใส่ boundary ให้

const data = await res.json();
ข้อผิดพลาดยอดนิยม: ใส่ headers: { "Content-Type": "multipart/form-data" } → server อ่านไฟล์ไม่ได้ เพราะขาด boundary

5. Frontend JS

ถ้าอาจารย์ให้ HTML/CSS มาแล้ว เหลือแค่ 4 อย่างนี้

render ตาราง

const tbody = document.querySelector("#books-body");

function render(rows) {
  tbody.innerHTML = rows.map(b => `
    <tr>
      <td><img src="${b.coverUrl || "/placeholder.png"}" width="46"></td>
      <td>${escapeHtml(b.title)}</td>
      <td>${escapeHtml(b.author)}</td>
      <td>${b.categoryName ?? ""}</td>
      <td>${b.price}</td>
      <td>
        <button data-edit="${b.id}">แก้ไข</button>
        <button data-del="${b.id}">ลบ</button>
      </td>
    </tr>`).join("");
}

โหลด + ลบ (event delegation)

async function load() {
  const res = await fetch("/api/books");
  const { data } = await res.json();
  render(data);
}
load();

document.addEventListener("click", async (e) => {
  const id = e.target.dataset.del;
  if (!id) return;
  if (!confirm("ลบเลยไหม?")) return;

  await fetch(`/api/books/${id}`, { method: "DELETE" });
  load();
});

ฟอร์ม + ไฟล์

form.addEventListener("submit", async (e) => {
  e.preventDefault();
  const fd = new FormData(form);

  const res = await fetch("/api/books", { method: "POST", body: fd });
  if (!res.ok) {
    const err = await res.json();
    alert(err.message);
    return;
  }
  form.reset();
  load();
});
อย่าลืม escapeHtml() ก่อนยัดค่าลง innerHTML — กัน XSS และกันหน้าพังเมื่อมีอักขระพิเศษ

6. Gen โค้ดจาก OpenAPI / Postman

ถ้าอาจารย์ให้ swagger.yaml/json มา อย่าเริ่มจากศูนย์

วิธี A — openapi-generator (เร็วกว่าเขียนเอง)

# ครั้งเดียว ไม่ต้องติดตั้ง global
npx @openapitools/openapi-generator-cli generate \
  -i openapi.yaml \
  -g nodejs-express-server \
  -o generated

# ตัวเลือก generator ที่ใช้บ่อย
#   nodejs-express-server  → Express + service stubs
#   javascript             → fetch client
#   typescript-fetch       → client + type
#   html2                  → เอกสาร API

จากนั้นแก้เฉพาะไฟล์ service ใน generated/services/ ให้ต่อ DB — routing/validation ได้มาฟรี

วิธี B — Postman

  1. Import openapi.yaml (หรือ swagger URL) เข้า Postman → ได้ collection ครบทุก endpoint
  2. ยิงทดสอบทีละตัวเพื่อดู response จริง
  3. กด Code snippet (ขวาบน) เลือก Node → Axios / fetch ได้โค้ดเรียก API ไปใช้ต่อ

วิธี C — อ่าน spec แล้วเขียนเอง (ชัวร์สุดถ้าเวลาน้อย)

  • ดู paths → ได้รายการ endpoint + method
  • ดู components.schemas → ได้ชื่อ field + type (เอาไปสร้างตาราง)
  • ดู requestBody ว่าเป็น multipart/form-data ไหม → รู้ว่าต้องใช้ multer
# ตรวจ spec ว่าถูกต้องก่อน gen
npx @redocly/cli lint openapi.yaml

# เปิดเอกสารสวย ๆ ให้เพื่อนดู
npx @redocly/cli preview-docs openapi.yaml
